     1→"use client";
     2→
     3→import { useEffect, useState } from "react";
     4→import { useAppStore } from "@/lib/store";
     5→import { Button } from "@/components/ui/button";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Badge } from "@/components/ui/badge";
     8→import { Input } from "@/components/ui/input";
     9→import { Label } from "@/components/ui/label";
    10→import { Textarea } from "@/components/ui/textarea";
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    12→import { Separator } from "@/components/ui/separator";
    13→import {
    14→  Select,
    15→  SelectContent,
    16→  SelectItem,
    17→  SelectTrigger,
    18→  SelectValue,
    19→} from "@/components/ui/select";
    20→import {
    21→  Dialog,
    22→  DialogContent,
    23→  DialogHeader,
    24→  DialogTitle,
    25→} from "@/components/ui/dialog";
    26→import {
    27→  ArrowLeft,
    28→  Phone,
    29→  Calendar,
    30→  MapPin,
    31→  Clock,
    32→  User,
    33→  MessageSquare,
    34→  Lock,
    35→  Edit,
    36→  Save,
    37→  X,
    38→  Plus,
    39→  CheckCircle2,
    40→  XCircle,
    41→  Building2,
    42→  Trash2,
    43→} from "lucide-react";
    44→
    45→const PIPELINE_STAGES = [
    46→  "New",
    47→  "Contacted",
    48→  "Qualified",
    49→  "Visit Scheduled",
    50→  "Visited",
    51→  "Negotiation",
    52→  "Won",
    53→  "Lost",
    54→];
    55→
    56→const statusColors: Record<string, string> = {
    57→  New: "bg-slate-500 text-white",
    58→  Contacted: "bg-blue-500 text-white",
    59→  Qualified: "bg-cyan-500 text-white",
    60→  "Visit Scheduled": "bg-amber-500 text-white",
    61→  Visited: "bg-purple-500 text-white",
    62→  Negotiation: "bg-orange-500 text-white",
    63→  Won: "bg-emerald-500 text-white",
    64→  Lost: "bg-red-500 text-white",
    65→};
    66→
    67→interface LeadProperty {
    68→  id: string;
    69→  property: {
    70→    id: string;
    71→    name: string;
    72→    type: string;
    73→    status: string;
    74→    price: number | null;
    75→    location: string | null;
    76→    size: string | null;
    77→  };
    78→  createdAt: string;
    79→}
    80→
    81→interface LeadDetail {
    82→  id: string;
    83→  name: string;
    84→  phone: string;
    85→  email: string | null;
    86→  source: string;
    87→  budget: string | null;
    88→  notes: string | null;
    89→  pipelineStatus: string;
    90→  lostReason: string | null;
    91→  primaryOwnerId: string;
    92→  currentOwnerId: string;
    93→  primaryOwner: { id: string; name: string; email: string; role: string };
    94→  currentOwner: { id: string; name: string; email: string; role: string };
    95→  project: { id: string; name: string } | null;
    96→  leadProperties: LeadProperty[];
    97→  callLogs: Array<{
    98→    id: string;
    99→    notes: string;
   100→    callType: string;
   101→    callDate: string;
   102→    createdAt: string;
   103→    user: { id: string; name: string };
   104→  }>;
   105→  followUps: Array<{
   106→    id: string;
   107→    scheduledAt: string;
   108→    notes: string;
   109→    completed: boolean;
   110→    completedAt: string | null;
   111→    user: { id: string; name: string };
   112→  }>;
   113→  siteVisits: Array<{
   114→    id: string;
   115→    scheduledAt: string;
   116→    notes: string;
   117→    status: string;
   118→    feedback: string | null;
   119→    user: { id: string; name: string };
   120→  }>;
   121→  assignments: Array<{
   122→    id: string;
   123→    fromUser: { name: string };
   124→    toUser: { name: string };
   125→    reason: string | null;
   126→    createdAt: string;
   127→  }>;
   128→  timeline: Array<{
   129→    id: string;
   130→    eventType: string;
   131→    description: string;
   132→    createdAt: string;
   133→    user: { name: string } | null;
   134→  }>;
   135→  createdAt: string;
   136→  updatedAt: string;
   137→}
   138→
   139→export function LeadDetail() {
   140→  const { user, selectedLeadId, setPage } = useAppStore();
   141→  const [lead, setLead] = useState<LeadDetail | null>(null);
   142→  const [loading, setLoading] = useState(true);
   143→  const [editing, setEditing] = useState(false);
   144→  const [editForm, setEditForm] = useState<Record<string, string>>({});
   145→
   146→  // Dialogs
   147→  const [callLogOpen, setCallLogOpen] = useState(false);
   148→  const [followUpOpen, setFollowUpOpen] = useState(false);
   149→  const [siteVisitOpen, setSiteVisitOpen] = useState(false);
   150→  const [lostReasonOpen, setLostReasonOpen] = useState(false);
   151→  const [lostReason, setLostReason] = useState("");
   152→
   153→  // Projects
   154→  const [addProjectOpen, setAddProjectOpen] = useState(false);
   155→  const [availableProjects, setAvailableProjects] = useState<Array<{
   156→    id: string;
   157→    name: string;
   158→    location?: string | null;
   159→  }>>([]);
   160→  const [selectedProjectId, setSelectedProjectId] = useState("");
   161→
   162→  const [users, setUsers] = useState<Array<{ id: string; name: string; role: string }>>([]);
   163→  const [sources, setSources] = useState<string[]>([]);
   164→
   165→  const fetchLead = async () => {
   166→    if (!selectedLeadId) return;
   167→    setLoading(true);
   168→    const res = await fetch(`/api/leads/${selectedLeadId}`);
   169→    if (res.ok) {
   170→      const data = await res.json();
   171→      setLead(data);
   172→      setEditForm({
   173→        name: data.name,
   174→        phone: data.phone,
   175→        email: data.email || "",
   176→        source: data.source,
   177→        budget: data.budget || "",
   178→        notes: data.notes || "",
   179→        pipelineStatus: data.pipelineStatus,
   180→      });
   181→    }
   182→    setLoading(false);
   183→  };
   184→
   185→  useEffect(() => {
   186→    let cancelled = false;
   187→    (async () => {
   188→      if (!selectedLeadId) {
   189→        setLoading(false);
   190→        return;
   191→      }
   192→      setLoading(true);
   193→      const res = await fetch(`/api/leads/${selectedLeadId}`);
   194→      if (!cancelled && res.ok) {
   195→        const data = await res.json();
   196→        setLead(data);
   197→        setEditForm({
   198→          name: data.name,
   199→          phone: data.phone,
   200→          email: data.email || "",
   201→          source: data.source,
   202→          budget: data.budget || "",
   203→          notes: data.notes || "",
   204→          pipelineStatus: data.pipelineStatus,
   205→        });
   206→      }
   207→      if (!cancelled) setLoading(false);
   208→
   209→      const uRes = await fetch("/api/users");
   210→      if (!cancelled && uRes.ok) setUsers(await uRes.json());
   211→
   212→      // Fetch lead sources for the dropdown
   213→      const srcRes = await fetch("/api/lead-sources");
   214→      if (!cancelled && srcRes.ok) {
   215→        const srcData = await srcRes.json();
   216→        if (Array.isArray(srcData)) {
   217→          setSources(srcData.map((s: { name: string }) => s.name));
   218→        }
   219→      }
   220→    })();
   221→    return () => { cancelled = true; };
   222→  }, [selectedLeadId]);
   223→
   224→  const canEdit =
   225→    user?.role === "admin" || user?.role === "super_admin" || lead?.currentOwnerId === user?.id;
   226→
   227→  const isViewOnly =
   228→    lead &&
   229→    user?.role === "sales" &&
   230→    lead.primaryOwnerId === user.id &&
   231→    lead.currentOwnerId !== user.id;
   232→
   233→  const handleSaveEdit = async () => {
   234→    if (!lead) return;
   235→    const res = await fetch(`/api/leads/${lead.id}`, {
   236→      method: "PUT",
   237→      headers: { "Content-Type": "application/json" },
   238→      body: JSON.stringify(editForm),
   239→    });
   240→    if (res.ok) {
   241→      setEditing(false);
   242→      fetchLead();
   243→    }
   244→  };
   245→
   246→  const handleStatusChange = async (status: string) => {
   247→    if (!lead || !canEdit) return;
   248→    if (status === "Lost") {
   249→      setLostReasonOpen(true);
   250→      return;
   251→    }
   252→    const res = await fetch(`/api/leads/${lead.id}`, {
   253→      method: "PUT",
   254→      headers: { "Content-Type": "application/json" },
   255→      body: JSON.stringify({ pipelineStatus: status }),
   256→    });
   257→    if (res.ok) fetchLead();
   258→  };
   259→
   260→  const handleMarkLost = async () => {
   261→    if (!lead) return;
   262→    const res = await fetch(`/api/leads/${lead.id}`, {
   263→      method: "PUT",
   264→      headers: { "Content-Type": "application/json" },
   265→      body: JSON.stringify({ pipelineStatus: "Lost", lostReason }),
   266→    });
   267→    if (res.ok) {
   268→      setLostReasonOpen(false);
   269→      setLostReason("");
   270→      fetchLead();
   271→    }
   272→  };
   273→
   274→  const handleLogCall = async (notes: string, callType: string, followUpDate?: string, dropLead?: boolean, leadStatus?: string, subStage?: string) => {
   275→    if (!lead) return;
   276→    // Send notes, callType, leadStatus AND subStage together.
   277→    // The /api/leads/[id]/call-logs route will:
   278→    //   1. create the CallLog with subStage
   279→    //   2. update the Lead's leadStatus + subStage in the same transaction
   280→    const res = await fetch(`/api/leads/${lead.id}/call-logs`, {
   281→      method: "POST",
   282→      headers: { "Content-Type": "application/json" },
   283→      body: JSON.stringify({
   284→        notes,
   285→        callType,
   286→        leadStatus: leadStatus || undefined,
   287→        subStage: subStage || undefined,
   288→      }),
   289→    });
   290→    if (res.ok) {
   291→      // If follow-up date set, schedule it
   292→      if (followUpDate) {
   293→        await fetch(`/api/leads/${lead.id}/follow-ups`, {
   294→          method: "POST",
   295→          headers: { "Content-Type": "application/json" },
   296→          body: JSON.stringify({
   297→            scheduledAt: followUpDate,
   298→            notes: `Follow-up after call: ${notes.substring(0, 100)}`,
   299→          }),
   300→        });
   301→      }
   302→
   303→      // If drop lead selected
   304→      if (dropLead) {
   305→        await fetch(`/api/leads/${lead.id}`, {
   306→          method: "PUT",
   307→          headers: { "Content-Type": "application/json" },
   308→          body: JSON.stringify({
   309→            pipelineStatus: "Lost",
   310→            lostReason: "Dropped after feedback",
   311→          }),
   312→        });
   313→      }
   314→
   315→      setCallLogOpen(false);
   316→      fetchLead();
   317→    }
   318→  };
   319→
   320→  const handleScheduleFollowUp = async (
   321→    scheduledAt: string,
   322→    notes: string
   323→  ) => {
   324→    if (!lead) return;
   325→    const res = await fetch(`/api/leads/${lead.id}/follow-ups`, {
   326→      method: "POST",
   327→      headers: { "Content-Type": "application/json" },
   328→      body: JSON.stringify({ scheduledAt, notes }),
   329→    });
   330→    if (res.ok) {
   331→      setFollowUpOpen(false);
   332→      fetchLead();
   333→    }
   334→  };
   335→
   336→  const handleScheduleSiteVisit = async (
   337→    scheduledAt: string,
   338→    notes: string
   339→  ) => {
   340→    if (!lead) return;
   341→    const res = await fetch(`/api/leads/${lead.id}/site-visits`, {
   342→      method: "POST",
   343→      headers: { "Content-Type": "application/json" },
   344→      body: JSON.stringify({ scheduledAt, notes }),
   345→    });
   346→    if (res.ok) {
   347→      setSiteVisitOpen(false);
   348→      fetchLead();
   349→    }
   350→  };
   351→
   352→  // Project management
   353→  const handleAddProject = async () => {
   354→    if (!lead || !selectedProjectId) return;
   355→    const res = await fetch(`/api/leads/${lead.id}`, {
   356→      method: "PUT",
   357→      headers: { "Content-Type": "application/json" },
   358→      body: JSON.stringify({ projectId: selectedProjectId }),
   359→    });
   360→    if (res.ok) {
   361→      setAddProjectOpen(false);
   362→      setSelectedProjectId("");
   363→      fetchLead();
   364→    }
   365→  };
   366→
   367→  const handleRemoveProject = async () => {
   368→    if (!lead) return;
   369→    const res = await fetch(`/api/leads/${lead.id}`, {
   370→      method: "PUT",
   371→      headers: { "Content-Type": "application/json" },
   372→      body: JSON.stringify({ projectId: null }),
   373→    });
   374→    if (res.ok) {
   375→      fetchLead();
   376→    }
   377→  };
   378→
   379→  const fetchAvailableProjects = async () => {
   380→    const res = await fetch("/api/projects");
   381→    if (res.ok) {
   382→      const data = await res.json();
   383→      setAvailableProjects(data);
   384→    }
   385→  };
   386→
   387→  if (loading) {
   388→    return (
   389→      <div className="space-y-4">
   390→        <div className="h-10 w-32 animate-pulse rounded bg-muted" />
   391→        <div className="h-48 animate-pulse rounded-lg bg-muted" />
   392→      </div>
   393→    );
   394→  }
   395→
   396→  if (!lead) {
   397→    return (
   398→      <div className="text-center py-12">
   399→        <p className="text-muted-foreground">Lead not found</p>
   400→        <Button variant="outline" className="mt-4" onClick={() => setPage("leads")}>
   401→          Back to Leads
   402→        </Button>
   403→      </div>
   404→    );
   405→  }
   406→
   407→  return (
   408→    <div className="space-y-4">
   409→      {/* Header */}
   410→      <div className="flex items-center gap-3">
   411→        <Button
   412→          variant="ghost"
   413→          size="sm"
   414→          onClick={() => setPage("leads")}
   415→        >
   416→          <ArrowLeft className="mr-1 h-4 w-4" /> Back
   417→        </Button>
   418→        <div className="flex-1">
   419→          <h2 className="text-xl font-bold text-foreground">{lead.name}</h2>
   420→          <p className="text-sm text-muted-foreground">{lead.phone}</p>
   421→        </div>
   422→        {canEdit && !editing && (
   423→          <Button
   424→            variant="outline"
   425→            size="sm"
   426→            onClick={() => setEditing(true)}
   427→          >
   428→            <Edit className="mr-1 h-3 w-3" /> Edit
   429→          </Button>
   430→        )}
   431→        {editing && (
   432→          <div className="flex gap-2">
   433→            <Button size="sm" onClick={handleSaveEdit} className="bg-brand hover:bg-brand-dark">
   434→              <Save className="mr-1 h-3 w-3" /> Save
   435→            </Button>
   436→            <Button
   437→              variant="outline"
   438→              size="sm"
   439→              onClick={() => setEditing(false)}
   440→            >
   441→              <X className="mr-1 h-3 w-3" /> Cancel
   442→            </Button>
   443→          </div>
   444→        )}
   445→      </div>
   446→
   447→      {/* View-only banner */}
   448→      {isViewOnly && (
   449→        <div className="flex items-center gap-2 rounded-lg bg-amber-50 dark:bg-amber-950 border border-amber-200 dark:border-amber-800 p-3">
   450→          <Lock className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   451→          <span className="text-sm text-amber-700 dark:text-amber-300">
   452→            Read-only — This lead is now with {lead.currentOwner.name}
   453→          </span>
   454→        </div>
   455→      )}
   456→
   457→      {/* Pipeline Status Bar */}
   458→      <Card>
   459→        <CardContent className="p-4">
   460→          <div className="flex items-center gap-1 overflow-x-auto pb-1">
   461→            {PIPELINE_STAGES.map((stage) => (
   462→              <button
   463→                key={stage}
   464→                onClick={() => canEdit && handleStatusChange(stage)}
   465→                disabled={!canEdit}
   466→                className={`flex items-center gap-1 whitespace-nowrap rounded-full px-3 py-1.5 text-xs font-medium transition-all ${
   467→                  lead.pipelineStatus === stage
   468→                    ? `${statusColors[stage]} ring-2 ring-offset-1 ring-brand`
   469→                    : "bg-muted text-muted-foreground hover:bg-accent"
   470→                } ${!canEdit ? "cursor-not-allowed opacity-60" : "cursor-pointer"}`}
   471→              >
   472→                {stage}
   473→              </button>
   474→            ))}
   475→          </div>
   476→          {lead.lostReason && (
   477→            <div className="mt-2 text-sm text-red-600 dark:text-red-400">
   478→              Lost Reason: {lead.lostReason}
   479→            </div>
   480→          )}
   481→        </CardContent>
   482→      </Card>
   483→
   484→      <div className="grid gap-4 lg:grid-cols-3">
   485→        {/* Lead Info */}
   486→        <div className="lg:col-span-2 space-y-4">
   487→          <Card>
   488→            <CardHeader className="pb-3">
   489→              <CardTitle className="text-base">Lead Information</CardTitle>
   490→            </CardHeader>
   491→            <CardContent>
   492→              {editing ? (
   493→                <div className="grid gap-3 sm:grid-cols-2">
   494→                  <div className="space-y-1">
   495→                    <Label>Name</Label>
   496→                    <Input
   497→                      value={editForm.name}
   498→                      onChange={(e) =>
   499→                        setEditForm({ ...editForm, name: e.target.value })
   500→                      }
   501→                    />
   502→                  </div>
   503→                  <div className="space-y-1">
   504→                    <Label>Phone</Label>
   505→                    <Input
   506→                      value={editForm.phone}
   507→                      onChange={(e) =>
   508→                        setEditForm({ ...editForm, phone: e.target.value })
   509→                      }
   510→                    />
   511→                  </div>
   512→                  <div className="space-y-1">
   513→                    <Label>Email</Label>
   514→                    <Input
   515→                      value={editForm.email}
   516→                      onChange={(e) =>
   517→                        setEditForm({ ...editForm, email: e.target.value })
   518→                      }
   519→                    />
   520→                  </div>
   521→                  <div className="space-y-1">
   522→                    <Label>Budget</Label>
   523→                    <Input
   524→                      value={editForm.budget}
   525→                      onChange={(e) =>
   526→                        setEditForm({ ...editForm, budget: e.target.value })
   527→                      }
   528→                    />
   529→                  </div>
   530→                  <div className="space-y-1">
   531→                    <Label>Source</Label>
   532→                    <Select
   533→                      value={editForm.source}
   534→                      onValueChange={(v) =>
   535→                        setEditForm({ ...editForm, source: v })
   536→                      }
   537→                    >
   538→                      <SelectTrigger>
   539→                        <SelectValue />
   540→                      </SelectTrigger>
   541→                      <SelectContent>
   542→                        {sources.map((s) => (
   543→                          <SelectItem key={s} value={s}>
   544→                            {s}
   545→                          </SelectItem>
   546→                        ))}
   547→                      </SelectContent>
   548→                    </Select>
   549→                  </div>
   550→                  <div className="sm:col-span-2 space-y-1">
   551→                    <Label>Notes</Label>
   552→                    <Textarea
   553→                      value={editForm.notes}
   554→                      onChange={(e) =>
   555→                        setEditForm({ ...editForm, notes: e.target.value })
   556→                      }
   557→                      rows={3}
   558→                    />
   559→                  </div>
   560→                </div>
   561→              ) : (
   562→                <div className="grid gap-3 sm:grid-cols-2">
   563→                  <div>
   564→                    <div className="text-xs text-muted-foreground">Email</div>
   565→                    <div className="text-sm text-foreground">{lead.email || "—"}</div>
   566→                  </div>
   567→                  <div>
   568→                    <div className="text-xs text-muted-foreground">Source</div>
   569→                    <div className="text-sm text-foreground">{lead.source}</div>
   570→                  </div>
   571→                  <div>
   572→                    <div className="text-xs text-muted-foreground">Lead Status</div>
   573→                    <div className="text-sm text-foreground">
   574→                      {(lead as Record<string, unknown>).leadStatus && (lead as Record<string, unknown>).leadStatus !== "New"
   575→                        ? (lead as Record<string, unknown>).leadStatus as string
   576→                        : "—"}
   577→                      {(lead as Record<string, unknown>).subStage && (
   578→                        <span className="ml-2 inline-flex items-center px-1.5 py-0.5 rounded text-[10px] font-medium bg-muted text-muted-foreground">
   579→                          {(lead as Record<string, unknown>).subStage as string}
   580→                        </span>
   581→                      )}
   582→                    </div>
   583→                  </div>
   584→                  <div>
   585→                    <div className="text-xs text-muted-foreground">Budget</div>
   586→                    <div className="text-sm text-foreground">{lead.budget || "—"}</div>
   587→                  </div>
   588→                  <div>
   589→                    <div className="text-xs text-muted-foreground">Project</div>
   590→                    <div className="text-sm text-foreground">
   591→                      {lead.project?.name || "—"}
   592→                    </div>
   593→                  </div>
   594→                  <div className="sm:col-span-2">
   595→                    <div className="text-xs text-muted-foreground">Notes</div>
   596→                    <div className="text-sm text-foreground">{lead.notes || "—"}</div>
   597→                  </div>
   598→                </div>
   599→              )}
   600→            </CardContent>
   601→          </Card>
   602→
   603→          {/* Project */}
   604→          <Card>
   605→            <CardHeader className="flex flex-row items-center justify-between pb-3">
   606→              <CardTitle className="text-base flex items-center gap-2">
   607→                <Building2 className="h-4 w-4" />
   608→                Project
   609→              </CardTitle>
   610→              <Button
   611→                variant="outline"
   612→                size="sm"
   613→                onClick={() => {
   614→                  fetchAvailableProjects();
   615→                  setAddProjectOpen(true);
   616→                }}
   617→              >
   618→                <Plus className="mr-1 h-3 w-3" /> {lead.project ? "Change Project" : "Set Project"}
   619→              </Button>
   620→            </CardHeader>
   621→            <CardContent>
   622→              {lead.project ? (

  1031→// Sub-forms
