   112→   112→function getDateRange(preset: number): { from: string; to: string } {
   113→   113→  const now = new Date();
   114→   114→  const to = now.toISOString().split("T")[0];
   115→   115→  let from: string;
   116→   116→
   117→   117→  if (preset === -1) {
   118→   118→    // This Month
   119→   119→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   120→   120→  } else if (preset === -2) {
   121→   121→    // Last Month
   122→   122→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   123→   123→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   124→   124→    from = firstDayLastMonth.toISOString().split("T")[0];
   125→   125→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   126→   126→  } else {
   127→   127→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   128→   128→  }
   129→   129→  return { from, to };
   130→   130→}

   192→   192→    const url = window.URL.createObjectURL(blob);
   193→   193→    const a = document.createElement("a");
   194→   194→    a.href = url;
   195→   195→    a.download = `${type}-export.csv`;
   196→   196→    a.click();
