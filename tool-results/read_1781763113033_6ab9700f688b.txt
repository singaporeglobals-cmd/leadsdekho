    85→function getDateRange(preset: number): { from: string; to: string } {
    86→  const now = new Date();
    87→  const to = now.toISOString().split("T")[0];
    88→  let from: string;
    89→  if (preset === -1) {
    90→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
    91→  } else if (preset === -2) {
    92→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
    93→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
    94→    from = firstDayLastMonth.toISOString().split("T")[0];
    95→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
    96→  } else {
    97→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
    98→  }
    99→  return { from, to };
   100→}
