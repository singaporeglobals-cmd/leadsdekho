   138→   138→
   139→   139→export function LeadDetail() {
   140→   140→  const { user, selectedLeadId, setPage } = useAppStore();
   141→   141→  const [lead, setLead] = useState<LeadDetail | null>(null);
   142→   142→  const [loading, setLoading] = useState(true);
   143→   143→  const [editing, setEditing] = useState(false);
   144→   144→  const [editForm, setEditForm] = useState<Record<string, string>>({});
   145→   145→
   146→   146→  // Dialogs
   147→   147→  const [callLogOpen, setCallLogOpen] = useState(false);
   148→   148→  const [followUpOpen, setFollowUpOpen] = useState(false);
   149→   149→  const [siteVisitOpen, setSiteVisitOpen] = useState(false);
   150→   150→  const [lostReasonOpen, setLostReasonOpen] = useState(false);
   151→   151→  const [lostReason, setLostReason] = useState("");
   152→   152→
   153→   153→  // Projects
   154→   154→  const [addProjectOpen, setAddProjectOpen] = useState(false);
   155→   155→  const [availableProjects, setAvailableProjects] = useState<Array<{
   156→   156→    id: string;
   157→   157→    name: string;
   158→   158→    location?: string | null;
   159→   159→  }>>([]);
   160→   160→  const [selectedProjectId, setSelectedProjectId] = useState("");
   161→   161→
   162→   162→  const [users, setUsers] = useState<Array<{ id: string; name: string; role: string }>>([]);
   163→   163→  const [sources, setSources] = useState<string[]>([]);
   164→   164→
   165→   165→  const fetchLead = async () => {
   166→   166→    if (!selectedLeadId) return;
   167→   167→    setLoading(true);
   168→   168→    const res = await fetch(`/api/leads/${selectedLeadId}`);
   169→   169→    if (res.ok) {
   170→   170→      const data = await res.json();
   171→   171→      setLead(data);
   172→   172→      setEditForm({
   173→   173→        name: data.name,
   174→   174→        phone: data.phone,
   175→   175→        email: data.email || "",
   176→   176→        source: data.source,
   177→   177→        budget: data.budget || "",
   178→   178→        notes: data.notes || "",
   179→   179→        pipelineStatus: data.pipelineStatus,
   180→   180→      });
   181→   181→    }
   182→   182→    setLoading(false);
   183→   183→  };
   184→   184→
   185→   185→  useEffect(() => {
   186→   186→    let cancelled = false;
   187→   187→    (async () => {
   188→   188→      if (!selectedLeadId) {
   189→   189→        setLoading(false);
   190→   190→        return;
   191→   191→      }
   192→   192→      setLoading(true);
   193→   193→      const res = await fetch(`/api/leads/${selectedLeadId}`);
   194→   194→      if (!cancelled && res.ok) {
   195→   195→        const data = await res.json();
   196→   196→        setLead(data);
   197→   197→        setEditForm({
   198→   198→          name: data.name,
   199→   199→          phone: data.phone,
   200→   200→          email: data.email || "",
   201→   201→          source: data.source,
   202→   202→          budget: data.budget || "",
   203→   203→          notes: data.notes || "",
   204→   204→          pipelineStatus: data.pipelineStatus,
   205→   205→        });
   206→   206→      }
   207→   207→      if (!cancelled) setLoading(false);
   208→   208→
   209→   209→      const uRes = await fetch("/api/users");
   210→   210→      if (!cancelled && uRes.ok) setUsers(await uRes.json());
   211→   211→
   212→   212→      // Fetch lead sources for the dropdown
   213→   213→      const srcRes = await fetch("/api/lead-sources");
   214→   214→      if (!cancelled && srcRes.ok) {
   215→   215→        const srcData = await srcRes.json();
   216→   216→        if (Array.isArray(srcData)) {
   217→   217→          setSources(srcData.map((s: { name: string }) => s.name));
   218→   218→        }
   219→   219→      }
   220→   220→    })();
   221→   221→    return () => { cancelled = true; };
   222→   222→  }, [selectedLeadId]);
   223→   223→
   224→   224→  const canEdit =
   225→   225→    user?.role === "admin" || user?.role === "super_admin" || lead?.currentOwnerId === user?.id;
   226→   226→
   227→   227→  const isViewOnly =
   228→   228→    lead &&
   229→   229→    user?.role === "sales" &&
   230→   230→    lead.primaryOwnerId === user.id &&
   231→   231→    lead.currentOwnerId !== user.id;
   232→   232→
   233→   233→  const handleSaveEdit = async () => {
   234→   234→    if (!lead) return;
   235→   235→    const res = await fetch(`/api/leads/${lead.id}`, {
   236→   236→      method: "PUT",
   237→   237→      headers: { "Content-Type": "application/json" },
   238→   238→      body: JSON.stringify(editForm),
   239→   239→    });
   240→   240→    if (res.ok) {
   241→   241→      setEditing(false);
   242→   242→      fetchLead();
   243→   243→    }
   244→   244→  };
   245→   245→
   246→   246→  const handleStatusChange = async (status: string) => {
   247→   247→    if (!lead || !canEdit) return;
   248→   248→    if (status === "Lost") {
   249→   249→      setLostReasonOpen(true);
   250→   250→      return;
   251→   251→    }
   252→   252→    const res = await fetch(`/api/leads/${lead.id}`, {
   253→   253→      method: "PUT",
   254→   254→      headers: { "Content-Type": "application/json" },
   255→   255→      body: JSON.stringify({ pipelineStatus: status }),
   256→   256→    });
   257→   257→    if (res.ok) fetchLead();
   258→   258→  };
   259→   259→
   260→   260→  const handleMarkLost = async () => {
   261→   261→    if (!lead) return;
   262→   262→    const res = await fetch(`/api/leads/${lead.id}`, {
   263→   263→      method: "PUT",
   264→   264→      headers: { "Content-Type": "application/json" },
   265→   265→      body: JSON.stringify({ pipelineStatus: "Lost", lostReason }),
   266→   266→    });
   267→   267→    if (res.ok) {
   268→   268→      setLostReasonOpen(false);
   269→   269→      setLostReason("");
   270→   270→      fetchLead();
   271→   271→    }
   272→   272→  };
   273→   273→
   274→   274→  const handleLogCall = async (notes: string, callType: string, followUpDate?: string, dropLead?: boolean, leadStatus?: string, subStage?: string) => {
   275→   275→    if (!lead) return;
   276→   276→    // Send notes, callType, leadStatus AND subStage together.
   277→   277→    // The /api/leads/[id]/call-logs route will:
   278→   278→    //   1. create the CallLog with subStage
   279→   279→    //   2. update the Lead's leadStatus + subStage in the same transaction
   280→   280→    const res = await fetch(`/api/leads/${lead.id}/call-logs`, {
   281→   281→      method: "POST",
   282→   282→      headers: { "Content-Type": "application/json" },
   283→   283→      body: JSON.stringify({
   284→   284→        notes,
   285→   285→        callType,
   286→   286→        leadStatus: leadStatus || undefined,
   287→   287→        subStage: subStage || undefined,
   288→   288→      }),
   289→   289→    });
   290→   290→    if (res.ok) {
   291→   291→      // If follow-up date set, schedule it
   292→   292→      if (followUpDate) {
   293→   293→        await fetch(`/api/leads/${lead.id}/follow-ups`, {
   294→   294→          method: "POST",
   295→   295→          headers: { "Content-Type": "application/json" },
   296→   296→          body: JSON.stringify({
   297→   297→            scheduledAt: followUpDate,
   298→   298→            notes: `Follow-up after call: ${notes.substring(0, 100)}`,
   299→   299→          }),
   300→   300→        });
   301→   301→      }
   302→   302→
   303→   303→      // If drop lead selected
   304→   304→      if (dropLead) {
   305→   305→        await fetch(`/api/leads/${lead.id}`, {
   306→   306→          method: "PUT",
   307→   307→          headers: { "Content-Type": "application/json" },
   308→   308→          body: JSON.stringify({
   309→   309→            pipelineStatus: "Lost",
   310→   310→            lostReason: "Dropped after feedback",
   311→   311→          }),
   312→   312→        });
   313→   313→      }
   314→   314→
   315→   315→      setCallLogOpen(false);
   316→   316→      fetchLead();
   317→   317→    }
   318→   318→  };
   319→   319→
   320→   320→  const handleScheduleFollowUp = async (
   321→   321→    scheduledAt: string,
   322→   322→    notes: string
   323→   323→  ) => {
   324→   324→    if (!lead) return;
   325→   325→    const res = await fetch(`/api/leads/${lead.id}/follow-ups`, {
   326→   326→      method: "POST",
   327→   327→      headers: { "Content-Type": "application/json" },
   328→   328→      body: JSON.stringify({ scheduledAt, notes }),
   329→   329→    });
   330→   330→    if (res.ok) {
   331→   331→      setFollowUpOpen(false);
   332→   332→      fetchLead();
   333→   333→    }
   334→   334→  };
   335→   335→
   336→   336→  const handleScheduleSiteVisit = async (
   337→   337→    scheduledAt: string,
   338→   338→    notes: string
   339→   339→  ) => {
   340→   340→    if (!lead) return;
   341→   341→    const res = await fetch(`/api/leads/${lead.id}/site-visits`, {
   342→   342→      method: "POST",
   343→   343→      headers: { "Content-Type": "application/json" },
   344→   344→      body: JSON.stringify({ scheduledAt, notes }),
   345→   345→    });
   346→   346→    if (res.ok) {
   347→   347→      setSiteVisitOpen(false);
   348→   348→      fetchLead();
   349→   349→    }
   350→   350→  };
   351→   351→
   352→   352→  // Project management
   353→   353→  const handleAddProject = async () => {
   354→   354→    if (!lead || !selectedProjectId) return;
   355→   355→    const res = await fetch(`/api/leads/${lead.id}`, {
   356→   356→      method: "PUT",
   357→   357→      headers: { "Content-Type": "application/json" },
   358→   358→      body: JSON.stringify({ projectId: selectedProjectId }),
   359→   359→    });
   360→   360→    if (res.ok) {
   361→   361→      setAddProjectOpen(false);
   362→   362→      setSelectedProjectId("");
   363→   363→      fetchLead();
   364→   364→    }
   365→   365→  };
   366→   366→
   367→   367→  const handleRemoveProject = async () => {
   368→   368→    if (!lead) return;
   369→   369→    const res = await fetch(`/api/leads/${lead.id}`, {
   370→   370→      method: "PUT",
   371→   371→      headers: { "Content-Type": "application/json" },
   372→   372→      body: JSON.stringify({ projectId: null }),
   373→   373→    });
   374→   374→    if (res.ok) {
   375→   375→      fetchLead();
   376→   376→    }
   377→   377→  };
   378→   378→
   379→   379→  const fetchAvailableProjects = async () => {
   380→   380→    const res = await fetch("/api/projects");
   381→   381→    if (res.ok) {
   382→   382→      const data = await res.json();
   383→   383→      setAvailableProjects(data);
   384→   384→    }
   385→   385→  };
   386→   386→
   387→   387→  if (loading) {
   388→   388→    return (
   389→   389→      <div className="space-y-4">
   390→   390→        <div className="h-10 w-32 animate-pulse rounded bg-muted" />
   391→   391→        <div className="h-48 animate-pulse rounded-lg bg-muted" />
   392→   392→      </div>
   393→   393→    );
   394→   394→  }
   395→   395→
   396→   396→  if (!lead) {
   397→   397→    return (
   398→   398→      <div className="text-center py-12">
   399→   399→        <p className="text-muted-foreground">Lead not found</p>
   400→   400→        <Button variant="outline" className="mt-4" onClick={() => setPage("leads")}>
   401→   401→          Back to Leads
   402→   402→        </Button>
   403→   403→      </div>
   404→   404→    );
   405→   405→  }
   406→   406→
   407→   407→  return (
   408→   408→    <div className="space-y-4">
   409→   409→      {/* Header */}
   410→   410→      <div className="flex items-center gap-3">
   411→   411→        <Button
   412→   412→          variant="ghost"
   413→   413→          size="sm"
   414→   414→          onClick={() => setPage("leads")}
   415→   415→        >
   416→   416→          <ArrowLeft className="mr-1 h-4 w-4" /> Back
   417→   417→        </Button>
   418→   418→        <div className="flex-1">
   419→   419→          <h2 className="text-xl font-bold text-foreground">{lead.name}</h2>
   420→   420→          <p className="text-sm text-muted-foreground">{lead.phone}</p>
   421→   421→        </div>
   422→   422→        {canEdit && !editing && (
   423→   423→          <Button
   424→   424→            variant="outline"
   425→   425→            size="sm"
   426→   426→            onClick={() => setEditing(true)}
   427→   427→          >
   428→   428→            <Edit className="mr-1 h-3 w-3" /> Edit
   429→   429→          </Button>
   430→   430→        )}
   431→   431→        {editing && (
   432→   432→          <div className="flex gap-2">
   433→   433→            <Button size="sm" onClick={handleSaveEdit} className="bg-brand hover:bg-brand-dark">
   434→   434→              <Save className="mr-1 h-3 w-3" /> Save
   435→   435→            </Button>
   436→   436→            <Button
   437→   437→              variant="outline"
   438→   438→              size="sm"
   439→   439→              onClick={() => setEditing(false)}
   440→   440→            >
   441→   441→              <X className="mr-1 h-3 w-3" /> Cancel
   442→   442→            </Button>
   443→   443→          </div>
   444→   444→        )}
   445→   445→      </div>
   446→   446→
   447→   447→      {/* View-only banner */}
   448→   448→      {isViewOnly && (
   449→   449→        <div className="flex items-center gap-2 rounded-lg bg-amber-50 dark:bg-amber-950 border border-amber-200 dark:border-amber-800 p-3">
   450→   450→          <Lock className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   451→   451→          <span className="text-sm text-amber-700 dark:text-amber-300">
   452→   452→            Read-only — This lead is now with {lead.currentOwner.name}
   453→   453→          </span>
   454→   454→        </div>
   455→   455→      )}
   456→   456→
   457→   457→      {/* Pipeline Status Bar */}
   458→   458→      <Card>
   459→   459→        <CardContent className="p-4">
   460→   460→          <div className="flex items-center gap-1 overflow-x-auto pb-1">
   461→   461→            {PIPELINE_STAGES.map((stage) => (
   462→   462→              <button
   463→   463→                key={stage}
   464→   464→                onClick={() => canEdit && handleStatusChange(stage)}
   465→   465→                disabled={!canEdit}
   466→   466→                className={`flex items-center gap-1 whitespace-nowrap rounded-full px-3 py-1.5 text-xs font-medium transition-all ${
   467→   467→                  lead.pipelineStatus === stage
   468→   468→                    ? `${statusColors[stage]} ring-2 ring-offset-1 ring-brand`
   469→   469→                    : "bg-muted text-muted-foreground hover:bg-accent"
   470→   470→                } ${!canEdit ? "cursor-not-allowed opacity-60" : "cursor-pointer"}`}
   471→   471→              >
   472→   472→                {stage}
   473→   473→              </button>
   474→   474→            ))}
   475→   475→          </div>
   476→   476→          {lead.lostReason && (
   477→   477→            <div className="mt-2 text-sm text-red-600 dark:text-red-400">
   478→   478→              Lost Reason: {lead.lostReason}
   479→   479→            </div>
   480→   480→          )}
   481→   481→        </CardContent>
   482→   482→      </Card>
   483→   483→
   484→   484→      <div className="grid gap-4 lg:grid-cols-3">
   485→   485→        {/* Lead Info */}
   486→   486→        <div className="lg:col-span-2 space-y-4">
   487→   487→          <Card>
   488→   488→            <CardHeader className="pb-3">
   489→   489→              <CardTitle className="text-base">Lead Information</CardTitle>
   490→   490→            </CardHeader>
   491→   491→            <CardContent>
   492→   492→              {editing ? (
   493→   493→                <div className="grid gap-3 sm:grid-cols-2">
   494→   494→                  <div className="space-y-1">
   495→   495→                    <Label>Name</Label>
   496→   496→                    <Input
   497→   497→                      value={editForm.name}
   498→   498→                      onChange={(e) =>
   499→   499→                        setEditForm({ ...editForm, name: e.target.value })
   500→   500→                      }
   501→   501→                    />
   502→   502→                  </div>
   503→   503→                  <div className="space-y-1">
   504→   504→                    <Label>Phone</Label>
   505→   505→                    <Input
   506→   506→                      value={editForm.phone}
   507→   507→                      onChange={(e) =>
   508→   508→                        setEditForm({ ...editForm, phone: e.target.value })
   509→   509→                      }
