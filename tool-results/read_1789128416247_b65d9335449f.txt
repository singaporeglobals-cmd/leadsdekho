   107→function formatDate(dateStr: string) {
   108→  if (!dateStr) return "";
   109→  const d = new Date(dateStr + "T00:00:00");
   110→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   111→}

   113→function getDateRange(preset: number): { from: string; to: string } {
   114→  const now = new Date();
   115→  const to = now.toISOString().split("T")[0];
   116→  let from: string;
   117→
   118→  if (preset === -1) {
   119→    // This Month
   120→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   121→  } else if (preset === -2) {
   122→    // Last Month
   123→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   124→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   125→    from = firstDayLastMonth.toISOString().split("T")[0];
   126→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   127→  } else {
   128→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   129→  }
   130→  return { from, to };
   131→}

   186→async function handleExport(type: string, from?: string, to?: string) {
   187→  const params = new URLSearchParams({ type });
   188→  if (from) params.set("from", from);
   189→  if (to) params.set("to", to);
   190→  const res = await fetch(`/api/reports/export?${params}`);
   191→  if (res.ok) {
   192→    const blob = await res.blob();
   193→    const url = window.URL.createObjectURL(blob);
   194→    const a = document.createElement("a");
   195→    a.href = url;
   196→    a.download = `${type}-export.csv`;
   197→    a.click();
   198→    window.URL.revokeObjectURL(url);
   199→  }
   200→}
