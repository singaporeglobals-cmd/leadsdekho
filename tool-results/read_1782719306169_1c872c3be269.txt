    47→    47→function generateMonthOptions() {
    48→    48→  const options = [{ value: "all", label: "All Time" }];
    49→    49→  const now = new Date();
    50→    50→  for (let i = 0; i < 12; i++) {
    51→    51→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    52→    52→    const value = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    53→    53→    const label = d.toLocaleDateString("en-US", { month: "short", year: "numeric" });
    54→    54→    options.push({ value, label });
    55→    55→  }
    56→    56→  return options;
    57→    57→}
