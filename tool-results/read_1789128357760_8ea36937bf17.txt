    48→function generateMonthOptions() {
    49→  const options = [{ value: "all", label: "All Time" }];
    50→  const now = new Date();
    51→  for (let i = 0; i < 12; i++) {
    52→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    53→    const value = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    54→    const label = d.toLocaleDateString("en-US", { month: "short", year: "numeric" });
    55→    options.push({ value, label });
    56→  }
    57→  return options;
    58→}
