   106→function formatDate(dateStr: string) {
   107→  if (!dateStr) return "";
   108→  const d = new Date(dateStr + "T00:00:00");
   109→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   110→}

   112→function getDateRange(preset: number): { from: string; to: string } {
   113→  const now = new Date();
   114→  const to = now.toISOString().split("T")[0];
   115→  let from: string;
   116→
   117→  if (preset === -1) {
   118→    // This Month
   119→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   120→  } else if (preset === -2) {
   121→    // Last Month
   122→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   123→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   124→    from = firstDayLastMonth.toISOString().split("T")[0];
   125→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   126→  } else {
   127→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   128→  }
   129→  return { from, to };
   130→}

   185→async function handleExport(type: string, from?: string, to?: string) {
   186→  const params = new URLSearchParams({ type });
   187→  if (from) params.set("from", from);
   188→  if (to) params.set("to", to);
   189→  const res = await fetch(`/api/reports/export?${params}`);
   190→  if (res.ok) {
   191→    const blob = await res.blob();
   192→    const url = window.URL.createObjectURL(blob);
   193→    const a = document.createElement("a");
   194→    a.href = url;
   195→    a.download = `${type}-export.csv`;
   196→    a.click();
   197→    window.URL.revokeObjectURL(url);
   198→  }
   199→}
