   115→   115→function getDateRange(preset: number): { from: string; to: string } {
   116→   116→  const now = new Date();
   117→   117→  const to = now.toISOString().split("T")[0];
   118→   118→  let from: string;
   119→   119→
   120→   120→  if (preset === -1) {
   121→   121→    // This Month
   122→   122→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   123→   123→  } else if (preset === -2) {
   124→   124→    // Last Month
   125→   125→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   126→   126→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   127→   127→    from = firstDayLastMonth.toISOString().split("T")[0];
   128→   128→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   129→   129→  } else {
   130→   130→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   131→   131→  }
   132→   132→  return { from, to };
   133→   133→}

   195→   195→    const url = window.URL.createObjectURL(blob);

   198→   198→    a.download = `${type}-export.csv`;
