    46→function generateMonthOptions() {
    47→  const options = [{ value: "all", label: "All Time" }];
    48→  const now = new Date();
    49→  for (let i = 0; i < 12; i++) {
    50→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    51→    const value = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    52→    const label = d.toLocaleDateString("en-US", { month: "short", year: "numeric" });
    53→    options.push({ value, label });
    54→  }
    55→  return options;
    56→}
