   109→function formatDate(dateStr: string) {
   110→  if (!dateStr) return "";
   111→  const d = new Date(dateStr + "T00:00:00");
   112→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" });
   113→}

   115→function getDateRange(preset: number): { from: string; to: string } {
   116→  const now = new Date();
   117→  const to = now.toISOString().split("T")[0];
   118→  let from: string;
   119→
   120→  if (preset === -1) {
   121→    // This Month
   122→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   123→  } else if (preset === -2) {
   124→    // Last Month
   125→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   126→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   127→    from = firstDayLastMonth.toISOString().split("T")[0];
   128→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   129→  } else {
   130→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   131→  }
   132→  return { from, to };
   133→}

   188→async function handleExport(type: string, from?: string, to?: string) {
   189→  const params = new URLSearchParams({ type });
   190→  if (from) params.set("from", from);
   191→  if (to) params.set("to", to);
   192→  const res = await fetch(`/api/reports/export?${params}`);
   193→  if (res.ok) {
   194→    const blob = await res.blob();
   195→    const url = window.URL.createObjectURL(blob);
   196→    const a = document.createElement("a");
   197→    a.href = url;
   198→    a.download = `${type}-export.csv`;
   199→    a.click();
   200→    window.URL.revokeObjectURL(url);
   201→  }
   202→}
