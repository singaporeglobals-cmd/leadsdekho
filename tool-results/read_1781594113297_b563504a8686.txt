    94→function getDateRange(preset: number): { from: string; to: string } {
    95→  const now = new Date();
    96→  const to = now.toISOString().split("T")[0];
    97→  let from: string;
    98→  if (preset === -1) {
    99→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   100→  } else if (preset === -2) {
   101→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   102→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   103→    from = firstDayLastMonth.toISOString().split("T")[0];
   104→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   105→  } else {
   106→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   107→  }
   108→  return { from, to };
   109→}
