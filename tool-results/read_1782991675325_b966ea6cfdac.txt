     1→     1→"use client";
     2→     2→
     3→     3→import { useState, useEffect, useCallback } from "react";
     4→     4→import { useAppStore } from "@/lib/store";
     5→     5→import { Button } from "@/components/ui/button";
     6→     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→     7→import { Input } from "@/components/ui/input";
     8→     8→import { Checkbox } from "@/components/ui/checkbox";
     9→     9→import {
    10→    10→  Select,
    11→    11→  SelectContent,
    12→    12→  SelectItem,
    13→    13→  SelectTrigger,
    14→    14→  SelectValue,
    15→    15→} from "@/components/ui/select";
    16→    16→import {
    17→    17→  Table,
    18→    18→  TableBody,
    19→    19→  TableCell,
    20→    20→  TableHead,
    21→    21→  TableHeader,
    22→    22→  TableRow,
    23→    23→} from "@/components/ui/table";
    24→    24→import { Badge } from "@/components/ui/badge";
    25→    25→import {
    26→    26→  RefreshCw,
    27→    27→  Building2,
    28→    28→  Users,
    29→    29→  Trash2,
    30→    30→  CheckCircle2,
    31→    31→  AlertTriangle,
    32→    32→  Inbox,
    33→    33→  Copy,
    34→    34→  Check,
    35→    35→} from "lucide-react";
    36→    36→
    37→    37→interface PortalLead {
    38→    38→  id: string;
    39→    39→  name: string;
    40→    40→  phone: string;
    41→    41→  email: string | null;
    42→    42→  source: string;
    43→    43→  budget: string | null;
    44→    44→  notes: string | null;
    45→    45→  projectName: string | null;
    46→    46→  portalRef: string | null;
    47→    47→  assignedTo: string | null;
    48→    48→  projectId: string | null;
    49→    49→  status: string;
    50→    50→  createdAt: string;
    51→    51→}
    52→    52→
    53→    53→interface UserItem {
    54→    54→  id: string;
    55→    55→  name: string;
    56→    56→  role: string;
    57→    57→  isActive: boolean;
    58→    58→}
    59→    59→
    60→    60→interface ProjectItem {
    61→    61→  id: string;
    62→    62→  name: string;
    63→    63→  location?: string | null;
    64→    64→}
    65→    65→
    66→    66→export function PortalLeads() {
    67→    67→  const { user } = useAppStore();
    68→    68→  const [portalLeads, setPortalLeads] = useState<PortalLead[]>([]);
    69→    69→  const [pendingCount, setPendingCount] = useState(0);
    70→    70→  const [loading, setLoading] = useState(true);
    71→    71→  const [error, setError] = useState("");
    72→    72→  const [users, setUsers] = useState<UserItem[]>([]);
    73→    73→  const [projects, setProjects] = useState<ProjectItem[]>([]);
    74→    74→  const [sources, setSources] = useState<string[]>([]);
    75→    75→  const [filterStatus, setFilterStatus] = useState<"pending" | "confirmed" | "discarded">("pending");
    76→    76→
    77→    77→  // Per-row editable fields (kept in a local map keyed by portal lead ID)
    78→    78→  const [rowAssign, setRowAssign] = useState<Record<string, string>>({});
    79→    79→  const [rowProject, setRowProject] = useState<Record<string, string>>({});
    80→    80→  const [rowSource, setRowSource] = useState<Record<string, string>>({});
    81→    81→
    82→    82→  // Bulk-apply state
    83→    83→  const [bulkProject, setBulkProject] = useState("");
    84→    84→  const [bulkAssign, setBulkAssign] = useState("");
    85→    85→  const [bulkSource, setBulkSource] = useState("");
    86→    86→  const [selectedRows, setSelectedRows] = useState<Set<string>>(new Set());
    87→    87→
    88→    88→  // Confirm / discard action state
    89→    89→  const [confirming, setConfirming] = useState(false);
    90→    90→  const [lastResult, setLastResult] = useState<{ confirmed: number; duplicated: number; failed: number } | null>(null);
    91→    91→
    92→    92→  // Copy API URL state
    93→    93→  const [copied, setCopied] = useState(false);
    94→    94→
    95→    95→  const apiUrl = `${typeof window !== "undefined" ? window.location.origin : ""}/api/portal-leads`;
    96→    96→
    97→    97→  const fetchPortalLeads = useCallback(async (status: "pending" | "confirmed" | "discarded") => {
    98→    98→    setLoading(true);
    99→    99→    setError("");
   100→   100→    try {
   101→   101→      const res = await fetch(`/api/portal-leads?status=${status}`);
   102→   102→      const data = await res.json();
   103→   103→      if (!res.ok) throw new Error(data.error || "Failed to fetch");
   104→   104→      setPortalLeads(data.portalLeads || []);
   105→   105→      setPendingCount(data.pendingCount || 0);
   106→   106→      // Reset per-row state for the freshly fetched rows
   107→   107→      const assignMap: Record<string, string> = {};
   108→   108→      const projectMap: Record<string, string> = {};
   109→   109→      const sourceMap: Record<string, string> = {};
   110→   110→      (data.portalLeads as PortalLead[]).forEach((p) => {
   111→   111→        if (p.assignedTo) assignMap[p.id] = p.assignedTo;
   112→   112→        if (p.projectId) projectMap[p.id] = p.projectId;
   113→   113→        sourceMap[p.id] = p.source;
   114→   114→      });
   115→   115→      setRowAssign(assignMap);
   116→   116→      setRowProject(projectMap);
   117→   117→      setRowSource(sourceMap);
   118→   118→      // Default-select all pending rows
   119→   119→      if (status === "pending") {
   120→   120→        setSelectedRows(new Set((data.portalLeads as PortalLead[]).map((p) => p.id)));
   121→   121→      } else {
   122→   122→        setSelectedRows(new Set());
   123→   123→      }
   124→   124→    } catch (err) {
   125→   125→      setError(err instanceof Error ? err.message : "Failed to fetch");
   126→   126→    } finally {
   127→   127→      setLoading(false);
   128→   128→    }
   129→   129→  }, []);
   130→   130→
   131→   131→  // Initial fetch
   132→   132→  useEffect(() => {
   133→   133→    if (user?.role === "admin" || user?.role === "super_admin") {
   134→   134→      fetchPortalLeads(filterStatus);
   135→   135→    }
   136→   136→  }, [user, filterStatus, fetchPortalLeads]);
   137→   137→
   138→   138→  // Fetch users + projects + sources
   139→   139→  useEffect(() => {
   140→   140→    fetch("/api/users").then((r) => r.json()).then(setUsers).catch(() => {});
   141→   141→    fetch("/api/projects").then((r) => r.json()).then(setProjects).catch(() => {});
   142→   142→    fetch("/api/lead-sources").then((r) => r.json()).then((data) => {
   143→   143→      if (Array.isArray(data)) setSources(data.map((s: { name: string }) => s.name));
   144→   144→    }).catch(() => {});
   145→   145→  }, []);
   146→   146→
   147→   147→  const toggleRow = (id: string) => {
   148→   148→    setSelectedRows((prev) => {
   149→   149→      const next = new Set(prev);
   150→   150→      if (next.has(id)) next.delete(id);
   151→   151→      else next.add(id);
   152→   152→      return next;
   153→   153→    });
   154→   154→  };
   155→   155→
   156→   156→  const toggleAll = () => {
   157→   157→    if (selectedRows.size === portalLeads.length && portalLeads.length > 0) {
   158→   158→      setSelectedRows(new Set());
   159→   159→    } else {
   160→   160→      setSelectedRows(new Set(portalLeads.map((p) => p.id)));
   161→   161→    }
   162→   162→  };
   163→   163→
   164→   164→  const applyBulkProject = () => {
   165→   165→    if (!bulkProject || selectedRows.size === 0) return;
   166→   166→    const map: Record<string, string> = { ...rowProject };
   167→   167→    selectedRows.forEach((id) => { map[id] = bulkProject; });
   168→   168→    setRowProject(map);
   169→   169→    setBulkProject("");
   170→   170→  };
   171→   171→
   172→   172→  const applyBulkAssign = () => {
   173→   173→    if (!bulkAssign || selectedRows.size === 0) return;
   174→   174→    const map: Record<string, string> = { ...rowAssign };
   175→   175→    selectedRows.forEach((id) => { map[id] = bulkAssign; });
   176→   176→    setRowAssign(map);
   177→   177→    setBulkAssign("");
   178→   178→  };
   179→   179→
   180→   180→  const applyBulkSource = () => {
   181→   181→    if (!bulkSource || selectedRows.size === 0) return;
   182→   182→    const map: Record<string, string> = { ...rowSource };
   183→   183→    selectedRows.forEach((id) => { map[id] = bulkSource; });
   184→   184→    setRowSource(map);
   185→   185→    setBulkSource("");
   186→   186→  };
   187→   187→
   188→   188→  const handleConfirm = async () => {
   189→   189→    if (selectedRows.size === 0) return;
   190→   190→    setConfirming(true);
   191→   191→    setLastResult(null);
   192→   192→    try {
   193→   193→      const payload = Array.from(selectedRows).map((id) => {
   194→   194→        const lead = portalLeads.find((p) => p.id === id);
   195→   195→        return {
   196→   196→          id,
   197→   197→          assignToId: rowAssign[id] || undefined,
   198→   198→          projectId: rowProject[id] || undefined,
   199→   199→          source: rowSource[id] || lead?.source || "Portal",
   200→   200→        };
   201→   201→      });
   202→   202→      const res = await fetch("/api/portal-leads/confirm", {
   203→   203→        method: "POST",
   204→   204→        headers: { "Content-Type": "application/json" },
   205→   205→        body: JSON.stringify({ leads: payload }),
   206→   206→      });
   207→   207→      const data = await res.json();
   208→   208→      if (!res.ok) throw new Error(data.error || "Confirm failed");
   209→   209→      setLastResult({
   210→   210→        confirmed: data.confirmed || 0,
   211→   211→        duplicated: data.duplicated || 0,
   212→   212→        failed: data.failed || 0,
   213→   213→      });
   214→   214→      // Refresh the list
   215→   215→      await fetchPortalLeads(filterStatus);
   216→   216→    } catch (err) {
   217→   217→      setError(err instanceof Error ? err.message : "Confirm failed");
   218→   218→    } finally {
   219→   219→      setConfirming(false);
   220→   220→    }
   221→   221→  };
   222→   222→
   223→   223→  const handleDiscard = async (id: string) => {
   224→   224→    if (!confirm("Discard this portal lead? You can find it later under the 'Discarded' filter.")) return;
   225→   225→    try {
   226→   226→      const res = await fetch(`/api/portal-leads/${id}`, { method: "DELETE" });
   227→   227→      if (!res.ok) {
   228→   228→        const data = await res.json();
   229→   229→        throw new Error(data.error || "Failed to discard");
   230→   230→      }
   231→   231→      await fetchPortalLeads(filterStatus);
   232→   232→    } catch (err) {
   233→   233→      setError(err instanceof Error ? err.message : "Failed to discard");
   234→   234→    }
   235→   235→  };
   236→   236→
   237→   237→  const handleDeleteForever = async (id: string) => {
   238→   238→    if (!confirm("PERMANENTLY delete this portal lead? This cannot be undone.")) return;
   239→   239→    try {
   240→   240→      const res = await fetch(`/api/portal-leads/${id}?hard=true`, { method: "DELETE" });
   241→   241→      if (!res.ok) {
   242→   242→        const data = await res.json();
   243→   243→        throw new Error(data.error || "Failed to delete");
   244→   244→      }
   245→   245→      await fetchPortalLeads(filterStatus);
   246→   246→    } catch (err) {
   247→   247→      setError(err instanceof Error ? err.message : "Failed to delete");
   248→   248→    }
   249→   249→  };
   250→   250→
   251→   251→  const copyApiUrl = () => {
   252→   252→    navigator.clipboard.writeText(apiUrl);
   253→   253→    setCopied(true);
   254→   254→    setTimeout(() => setCopied(false), 2000);
   255→   255→  };
   256→   256→
   257→   257→  if (user?.role !== "admin" && user?.role !== "super_admin") return null;
   258→   258→
   259→   259→  return (
   260→   260→    <div className="space-y-4">
   261→   261→      {/* API endpoint info card */}
   262→   262→      <Card>
   263→   263→        <CardHeader>
   264→   264→          <CardTitle className="text-lg flex items-center gap-2">
   265→   265→            <Inbox className="h-5 w-5" />
   266→   266→            Portal Leads — Pending Review
   267→   267→            {pendingCount > 0 && (
   268→   268→              <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-900 dark:text-amber-300 ml-2">
   269→   269→                {pendingCount} pending
   270→   270→              </Badge>
   271→   271→            )}
   272→   272→          </CardTitle>
   273→   273→        </CardHeader>
   274→   274→        <CardContent className="space-y-3">
   275→   275→          <div className="rounded-md bg-blue-50 dark:bg-blue-950 p-3 text-sm space-y-2">
   276→   276→            <div className="font-medium text-blue-700 dark:text-blue-300">
   277→   277→              External Portal API Endpoint
   278→   278→            </div>
   279→   279→            <div className="flex items-center gap-2">
   280→   280→              <code className="flex-1 px-2 py-1 bg-white dark:bg-gray-900 rounded border text-xs">
   281→   281→                POST {apiUrl}
   282→   282→              </code>
   283→   283→              <Button size="sm" variant="outline" onClick={copyApiUrl} className="h-8">
   284→   284→                {copied ? <Check className="h-3 w-3 mr-1" /> : <Copy className="h-3 w-3 mr-1" />}
   285→   285→                {copied ? "Copied" : "Copy"}
   286→   286→              </Button>
   287→   287→            </div>
   288→   288→            <p className="text-xs text-blue-700 dark:text-blue-300">
   289→   289→              Send leads from Housing.com, MagicBricks, 99acres, or any portal to this endpoint. Required fields: <code>name</code>, <code>phone</code>. Optional: <code>email</code>, <code>source</code>, <code>budget</code>, <code>notes</code>, <code>projectName</code>, <code>portalRef</code>.
   290→   290→            </p>
   291→   291→          </div>
   292→   292→
   293→   293→          {/* Filter tabs */}
   294→   294→          <div className="flex items-center gap-2">
   295→   295→            {(["pending", "confirmed", "discarded"] as const).map((s) => (
   296→   296→              <Button
   297→   297→                key={s}
   298→   298→                size="sm"
   299→   299→                variant={filterStatus === s ? "default" : "outline"}
   300→   300→                onClick={() => setFilterStatus(s)}
   301→   301→                className="h-8 capitalize"
   302→   302→              >
   303→   303→                {s}
   304→   304→                {s === "pending" && pendingCount > 0 && (
   305→   305→                  <span className="ml-1.5 px-1.5 py-0.5 bg-amber-500 text-white rounded-full text-xs">
   306→   306→                    {pendingCount}
   307→   307→                  </span>
   308→   308→                )}
   309→   309→              </Button>
   310→   310→            ))}
   311→   311→            <div className="flex-1" />
   312→   312→            <Button
   313→   313→              size="sm"
   314→   314→              variant="outline"
   315→   315→              onClick={() => fetchPortalLeads(filterStatus)}
   316→   316→              disabled={loading}
   317→   317→              className="h-8"
   318→   318→            >
   319→   319→              <RefreshCw className={`h-3 w-3 mr-1 ${loading ? "animate-spin" : ""}`} />
   320→   320→              Refresh
   321→   321→            </Button>
   322→   322→          </div>
   323→   323→
   324→   324→          {error && (
   325→   325→            <div className="rounded-md bg-red-50 dark:bg-red-950 p-3 text-sm text-red-600 dark:text-red-400">
   326→   326→              {error}
   327→   327→            </div>
   328→   328→          )}
   329→   329→
   330→   330→          {lastResult && (
   331→   331→            <div className="rounded-md bg-emerald-50 dark:bg-emerald-950 p-3 text-sm text-emerald-700 dark:text-emerald-300 flex items-center gap-2">
   332→   332→              <CheckCircle2 className="h-4 w-4" />
   333→   333→              <span>
   334→   334→                <strong>{lastResult.confirmed}</strong> lead(s) added to Lead Management
   335→   335→                {lastResult.duplicated > 0 && (
   336→   336→                  <span>, <strong>{lastResult.duplicated}</strong> skipped (duplicate phone)</span>
   337→   337→                )}
   338→   338→                {lastResult.failed > 0 && (
   339→   339→                  <span>, <strong>{lastResult.failed}</strong> failed</span>
   340→   340→                )}
   341→   341→              </span>
   342→   342→            </div>
   343→   343→          )}
   344→   344→
   345→   345→          {portalLeads.length === 0 ? (
   346→   346→            <div className="rounded-md border border-dashed p-8 text-center text-sm text-muted-foreground">
   347→   347→              No {filterStatus} portal leads.
   348→   348→              {filterStatus === "pending" && " When a portal sends a new lead, it will appear here."}
   349→   349→            </div>
   350→   350→          ) : (
   351→   351→            <>
   352→   352→              {/* Bulk-apply bar (only show in pending view) */}
   353→   353→              {filterStatus === "pending" && (
   354→   354→                <div className="space-y-3 p-3 bg-muted/50 rounded-lg border border-border">
   355→   355→                  <div className="flex items-center justify-between">
   356→   356→                    <div className="flex items-center gap-3">
   357→   357→                      <span className="text-sm font-medium">Selection:</span>
   358→   358→                      <Button size="sm" variant="outline" onClick={toggleAll} className="h-7 text-xs">
   359→   359→                        {selectedRows.size === portalLeads.length ? "Deselect All" : "Select All"}
   360→   360→                      </Button>
   361→   361→                      <span className="text-xs text-muted-foreground">
   362→   362→                        {selectedRows.size} of {portalLeads.length} selected
   363→   363→                      </span>
   364→   364→                    </div>
   365→   365→                  </div>
   366→   366→
   367→   367→                  <div className="flex flex-wrap gap-2 items-center">
   368→   368→                    <span className="text-sm font-medium text-brand">Bulk Apply to Selected:</span>
   369→   369→
   370→   370→                    <Select value={bulkProject} onValueChange={setBulkProject}>
   371→   371→                      <SelectTrigger className="w-[180px] h-8">
   372→   372→                        <Building2 className="mr-1 h-3 w-3 shrink-0" />
   373→   373→                        <SelectValue placeholder="Set Project" />
   374→   374→                      </SelectTrigger>
   375→   375→                      <SelectContent>
   376→   376→                        {projects.map((p) => (
   377→   377→                          <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   378→   378→                        ))}
   379→   379→                      </SelectContent>
   380→   380→                    </Select>
   381→   381→                    <Button size="sm" variant="outline" onClick={applyBulkProject} disabled={!bulkProject || selectedRows.size === 0} className="h-8 text-xs">
   382→   382→                      Apply Project
   383→   383→                    </Button>
   384→   384→
   385→   385→                    <Select value={bulkSource} onValueChange={setBulkSource}>
   386→   386→                      <SelectTrigger className="w-[150px] h-8">
   387→   387→                        <SelectValue placeholder="Set Source" />
   388→   388→                      </SelectTrigger>
   389→   389→                      <SelectContent>
   390→   390→                        {sources.map((s) => (
   391→   391→                          <SelectItem key={s} value={s}>{s}</SelectItem>
   392→   392→                        ))}
   393→   393→                      </SelectContent>
   394→   394→                    </Select>
   395→   395→                    <Button size="sm" variant="outline" onClick={applyBulkSource} disabled={!bulkSource || selectedRows.size === 0} className="h-8 text-xs">
   396→   396→                      Apply Source
   397→   397→                    </Button>
   398→   398→
   399→   399→                    <Select value={bulkAssign} onValueChange={setBulkAssign}>
   400→   400→                      <SelectTrigger className="w-[160px] h-8">
   401→   401→                        <Users className="mr-1 h-3 w-3 shrink-0" />
   402→   402→                        <SelectValue placeholder="Set Assignee" />
   403→   403→                      </SelectTrigger>
   404→   404→                      <SelectContent>
   405→   405→                        {users
   406→   406→                          .filter((u) => u.isActive && u.role !== "admin")
   407→   407→                          .map((u) => (
   408→   408→                            <SelectItem key={u.id} value={u.id}>
   409→   409→                              {u.name} ({u.role})
   410→   410→                            </SelectItem>
   411→   411→                          ))}
   412→   412→                      </SelectContent>
   413→   413→                    </Select>
   414→   414→                    <Button size="sm" variant="outline" onClick={applyBulkAssign} disabled={!bulkAssign || selectedRows.size === 0} className="h-8 text-xs">
   415→   415→                      Apply Assignee
   416→   416→                    </Button>
   417→   417→
   418→   418→                    <div className="flex-1" />
   419→   419→
   420→   420→                    <Button
   421→   421→                      onClick={handleConfirm}
   422→   422→                      disabled={confirming || selectedRows.size === 0}
   423→   423→                      className="h-8 text-xs bg-emerald-600 hover:bg-emerald-700"
   424→   424→                    >
   425→   425→                      <CheckCircle2 className="h-3 w-3 mr-1" />
   426→   426→                      {confirming ? "Confirming..." : `Confirm ${selectedRows.size} Lead${selectedRows.size !== 1 ? "s" : ""} → Lead Management`}
   427→   427→                    </Button>
   428→   428→                  </div>
   429→   429→                </div>
   430→   430→              )}
   431→   431→
   432→   432→              {/* Portal leads table */}
   433→   433→              <div className="max-h-[600px] overflow-auto rounded-lg border">
   434→   434→                <Table>
   435→   435→                  <TableHeader>
   436→   436→                    <TableRow>
   437→   437→                      {filterStatus === "pending" && (
   438→   438→                        <TableHead className="w-8">
   439→   439→                          <Checkbox
   440→   440→                            checked={selectedRows.size === portalLeads.length && portalLeads.length > 0}
   441→   441→                            onCheckedChange={toggleAll}
   442→   442→                          />
   443→   443→                        </TableHead>
   444→   444→                      )}
   445→   445→                      <TableHead className="w-8">#</TableHead>
   446→   446→                      <TableHead>Received</TableHead>
   447→   447→                      <TableHead>Source</TableHead>
   448→   448→                      <TableHead>Name</TableHead>
   449→   449→                      <TableHead>Phone</TableHead>
   450→   450→                      <TableHead>Email</TableHead>
   451→   451→                      <TableHead>Project (file)</TableHead>
   452→   452→                      <TableHead>Budget</TableHead>
