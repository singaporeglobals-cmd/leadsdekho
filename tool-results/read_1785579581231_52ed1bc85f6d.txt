    49→function generateMonthOptions() {
    50→  const options = [{ value: "all", label: "All Time" }];
    51→  const now = new Date();
    52→  for (let i = 0; i < 12; i++) {
    53→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    54→    const value = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    55→    const label = d.toLocaleDateString("en-US", { month: "short", year: "numeric" });
    56→    options.push({ value, label });
    57→  }
    58→  return options;
    59→}
