    93→    93→function getDateRange(preset: number): { from: string; to: string } {
    94→    94→  const now = new Date();
    95→    95→  const to = now.toISOString().split("T")[0];
    96→    96→  let from: string;
    97→    97→  if (preset === -1) {
    98→    98→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
    99→    99→  } else if (preset === -2) {
   100→   100→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   101→   101→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   102→   102→    from = firstDayLastMonth.toISOString().split("T")[0];
   103→   103→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   104→   104→  } else {
   105→   105→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   106→   106→  }
   107→   107→  return { from, to };
   108→   108→}
