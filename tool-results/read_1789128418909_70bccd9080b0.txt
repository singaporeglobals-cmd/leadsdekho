   113→   113→function getDateRange(preset: number): { from: string; to: string } {
   114→   114→  const now = new Date();
   115→   115→  const to = now.toISOString().split("T")[0];
   116→   116→  let from: string;
   117→   117→
   118→   118→  if (preset === -1) {
   119→   119→    // This Month
   120→   120→    from = new Date(now.getFullYear(), now.getMonth(), 1).toISOString().split("T")[0];
   121→   121→  } else if (preset === -2) {
   122→   122→    // Last Month
   123→   123→    const firstDayLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);
   124→   124→    const lastDayLastMonth = new Date(now.getFullYear(), now.getMonth(), 0);
   125→   125→    from = firstDayLastMonth.toISOString().split("T")[0];
   126→   126→    return { from, to: lastDayLastMonth.toISOString().split("T")[0] };
   127→   127→  } else {
   128→   128→    from = new Date(now.getTime() - preset * 24 * 60 * 60 * 1000).toISOString().split("T")[0];
   129→   129→  }
   130→   130→  return { from, to };
   131→   131→}
